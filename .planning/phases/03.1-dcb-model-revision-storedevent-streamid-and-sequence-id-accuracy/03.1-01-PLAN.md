**3. event-sourcing/src/query.rs** — New module:

```rust
use std::collections::HashSet;
use crate::types::Tag;
use crate::event::StoredEvent;

/// A single filter criterion. An event matches if ALL of the following hold:
/// - event_types is empty OR event.event_type is in event_types
/// - tags is empty OR event.tags is a superset of tags (all required tags present)
#[derive(Debug, Clone)]
pub struct Criterion {
    pub event_types: HashSet<String>,
    pub tags: HashSet<Tag>,
}

/// A query is a set of criteria. An event matches the query if it matches ANY criterion.
/// An empty criteria vec means "match nothing" -- use Query::all() for match-all.
#[derive(Debug, Clone)]
pub struct Query {
    pub criteria: Vec<Criterion>,
}

impl Query {
    /// Match all events.
    pub fn all() -> Self {
        Query {
            criteria: vec![Criterion {
                event_types: HashSet::new(),
                tags: HashSet::new(),
            }],
        }
    }

    /// Match events carrying ALL of the specified tags.
    pub fn match_tags(tags: impl IntoIterator<Item = Tag>) -> Self {
        Query {
            criteria: vec![Criterion {
                event_types: HashSet::new(),
                tags: tags.into_iter().collect(),
            }],
        }
    }

    /// Match events of any of the specified event types.
    pub fn match_event_types(event_types: impl IntoIterator<Item = impl Into<String>>) -> Self {
        Query {
            criteria: vec![Criterion {
                event_types: event_types.into_iter().map(Into::into).collect(),
                tags: HashSet::new(),
            }],
        }
    }

    /// Add an OR criterion to the query.
    pub fn or(mut self, criterion: Criterion) -> Self {
        self.criteria.push(criterion);
        self
    }
}

/// Check if a stored event matches a single criterion.
pub fn event_matches_criterion(event: &StoredEvent, criterion: &Criterion) -> bool {
    let type_match = criterion.event_types.is_empty()
        || criterion.event_types.contains(&event.event_type);
    let tag_match = criterion.tags.is_empty()
        || criterion.tags.iter().all(|t| event.tags.contains(t));
    type_match && tag_match
}

/// Check if a stored event matches a query (any criterion).
pub fn event_matches_query(event: &StoredEvent, query: &Query) -> bool {
    if query.criteria.is_empty() {
        return false;
    }
    query.criteria.iter().any(|c| event_matches_criterion(event, c))
}
```

Use `HashSet<String>` for `Criterion.event_types` (not Vec) for O(1) lookup per RESEARCH.md anti-pattern guidance.

Add unit tests:
- `query_all_matches_any_event`: Construct a StoredEvent, assert `event_matches_query` returns true with `Query::all()`
- `query_empty_criteria_matches_nothing`: `Query { criteria: vec![] }` matches nothing
- `query_match_tags_filters_correctly`: Event with tags `{"a", "b"}` matches `Query::match_tags([a])`, does NOT match `Query::match_tags([c])`
- `query_match_event_types_filters_correctly`: Event with type "X" matches `Query::match_event_types(["X"])`, not `Query::match_event_types(["Y"])`
- `query_criterion_and_semantics`: Criterion requiring both tag "a" AND event type "X" only matches events with BOTH
- `query_or_semantics`: Query with two criteria matches events matching either one
