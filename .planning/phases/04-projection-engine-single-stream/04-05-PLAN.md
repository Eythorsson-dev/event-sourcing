<interfaces>
From 04-01: `Event` trait, `EventSchemaDef`, `FieldDef`, `FieldType` (in `event_sourcing::schema`).

The derive macro generates this impl from a struct:
```rust
#[derive(Event)]
struct OrderPlaced {
    order_id: String,
    amount: f64,
    item_count: i32,
}
```
→ generates:
```rust
impl event_sourcing::Event for OrderPlaced {
    fn event_type() -> &'static str { "OrderPlaced" }
    fn schema() -> event_sourcing::EventSchemaDef {
        event_sourcing::EventSchemaDef {
            event_type: "OrderPlaced".to_string(),
            fields: vec![
                event_sourcing::FieldDef { name: "order_id".to_string(), field_type: event_sourcing::FieldType::String },
                event_sourcing::FieldDef { name: "amount".to_string(), field_type: event_sourcing::FieldType::Decimal },
                event_sourcing::FieldDef { name: "item_count".to_string(), field_type: event_sourcing::FieldType::Integer },
            ],
        }
    }
}
```

Type mapping (path-string matching — no type alias resolution, known limitation per A4):
| Rust type path | FieldType variant |
|----------------|-------------------|
| `String` / `std::string::String` | `FieldType::String` |
| `str` | `FieldType::String` |
| `i8` / `i16` / `i32` / `i64` / `i128` / `u8` / `u16` / `u32` / `u64` / `u128` / `usize` / `isize` | `FieldType::Integer` |
| `f32` / `f64` | `FieldType::Decimal` |
| `chrono::NaiveDate` | `FieldType::Date` |
| `chrono::DateTime` / `chrono::DateTime<chrono::Utc>` | `FieldType::DateTime` |
| `bool` | `FieldType::String` (stringified — v1 simplification) |
| anything else | compile error: `"unsupported field type for #[derive(Event)]: use String, integer, float, chrono::NaiveDate, or chrono::DateTime"` |

For `Option<T>` fields: unwrap the inner type and apply the mapping. The `Option` wrapper affects `required` in `ProjectionDefinition` (Plan 06) but not `FieldType` mapping here.
</interfaces>
</context>

    ```rust
    use proc_macro2::TokenStream;
    use quote::quote;
    use syn::{Data, DeriveInput, Error, Fields, Type, parse2};

    pub fn impl_derive_event(input: TokenStream) -> Result<TokenStream, Error> {
        let ast: DeriveInput = parse2(input)?;
        let name = &ast.ident;
        let name_str = name.to_string();

        let fields = match &ast.data {
            Data::Struct(s) => match &s.fields {
                Fields::Named(f) => &f.named,
                _ => return Err(Error::new_spanned(&ast, "#[derive(Event)] only supports structs with named fields")),
            },
            _ => return Err(Error::new_spanned(&ast, "#[derive(Event)] only supports structs")),
        };

        let field_defs: Vec<TokenStream> = fields
            .iter()
            .map(|f| {
                let field_name = f.ident.as_ref().unwrap().to_string();
                let field_type_tokens = map_rust_type_to_field_type(&f.ty, f)?;
                Ok(quote! {
                    event_sourcing::FieldDef {
                        name: #field_name.to_string(),
                        field_type: #field_type_tokens,
                    }
                })
            })
            .collect::<Result<Vec<_>, Error>>()?;

        Ok(quote! {
            impl event_sourcing::Event for #name {
                fn event_type() -> &'static str { #name_str }
                fn schema() -> event_sourcing::EventSchemaDef {
                    event_sourcing::EventSchemaDef {
                        event_type: #name_str.to_string(),
                        fields: vec![#(#field_defs),*],
                    }
                }
            }
        })
    }

    fn map_rust_type_to_field_type(ty: &Type, span_target: &impl quote::ToTokens) -> Result<TokenStream, Error> {
        let type_str = quote!(#ty).to_string().replace(" ", "");
        let inner_str = if type_str.starts_with("Option<") && type_str.ends_with('>') {
            &type_str[7..type_str.len() - 1]
        } else {
            type_str.as_str()
        };
        match inner_str {
            "String" | "std::string::String" | "&str" | "str" => Ok(quote! { event_sourcing::FieldType::String }),
            "i8" | "i16" | "i32" | "i64" | "i128" | "u8" | "u16" | "u32" | "u64" | "u128" | "usize" | "isize" => {
                Ok(quote! { event_sourcing::FieldType::Integer })
            }
            "f32" | "f64" => Ok(quote! { event_sourcing::FieldType::Decimal }),
            "bool" => Ok(quote! { event_sourcing::FieldType::String }),
            "chrono::NaiveDate" => Ok(quote! { event_sourcing::FieldType::Date }),
            "chrono::DateTime" | "chrono::DateTime<chrono::Utc>" | "DateTime<Utc>" => {
                Ok(quote! { event_sourcing::FieldType::DateTime })
            }
            _ => Err(Error::new_spanned(
                span_target,
                format!(
                    "unsupported field type for #[derive(Event)]: `{}`. Use String, integer primitive, f32/f64, chrono::NaiveDate, or chrono::DateTime",
                    inner_str
                ),
            )),
        }
    }
    ```
