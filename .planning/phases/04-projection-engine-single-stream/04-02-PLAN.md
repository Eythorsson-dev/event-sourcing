// event-sourcing/src/projection/definition.rs

use indexmap::IndexMap;
use serde_json::Value;

/// Handler for a single event on a scalar field.
#[derive(Debug, Clone, PartialEq, serde::Serialize, serde::Deserialize)]
#[serde(rename_all = "snake_case", deny_unknown_fields)]
pub enum HandlerSpec {
    /// Copy field from event payload: { "from": "$.path" }
    From { from: String },
    /// Set to a static literal or null: { "value": <literal> }
    Value { value: Value },
    /// Add N to a numeric field: { "increment": N }
    Increment { increment: f64 },
    /// Subtract N from a numeric field: { "decrement": N }
    Decrement { decrement: f64 },
    /// Add value at path to a numeric field: { "increment_by": "$.path" }
    IncrementBy { increment_by: String },
    /// Subtract value at path from a numeric field: { "decrement_by": "$.path" }
    DecrementBy { decrement_by: String },
}

/// Scalar field spec within a ProjectionDefinition.
#[derive(Debug, Clone, PartialEq, serde::Serialize, serde::Deserialize)]
#[serde(deny_unknown_fields)]
pub struct ScalarFieldSpec {
    /// If true, the Rust field type is T; if false (default), Option<T>.
    #[serde(default)]
    pub required: bool,
    /// Default value for initial state (before any events fire).
    #[serde(skip_serializing_if = "Option::is_none")]
    pub default: Option<Value>,
    /// Event type → handler mapping. Keys are bare event type names.
    pub events: IndexMap<String, HandlerSpec>,
}

/// Nested object field spec within a ProjectionDefinition.
#[derive(Debug, Clone, PartialEq, serde::Serialize, serde::Deserialize)]
#[serde(deny_unknown_fields)]
pub struct ObjectFieldSpec {
    /// Always "object".
    #[serde(rename = "type")]
    pub field_type: String,
    /// Event types that, when fired, clear the entire object to null.
    #[serde(default, skip_serializing_if = "Vec::is_empty")]
    pub cleared_by: Vec<String>,
    /// Nested scalar fields. Insertion order is preserved.
    pub fields: IndexMap<String, ScalarFieldSpec>,
}

/// A single top-level field — either a scalar or a nested object.
#[derive(Debug, Clone, PartialEq, serde::Serialize, serde::Deserialize)]
#[serde(untagged)]
pub enum FieldSpec {
    Scalar(ScalarFieldSpec),
    Object(ObjectFieldSpec),
}

/// The JSON-serializable description of what a projection computes.
#[derive(Debug, Clone, PartialEq, serde::Serialize, serde::Deserialize)]
#[serde(deny_unknown_fields)]
pub struct ProjectionDefinition {
    pub name: String,
    pub query: crate::query::TagFilter,
    /// Top-level fields. IndexMap preserves insertion order for deterministic JSON.
    pub fields: IndexMap<String, FieldSpec>,
}

impl ProjectionDefinition {
    pub fn builder(name: impl Into<String>, query: crate::query::TagFilter) -> ProjectionDefinitionBuilder {
        ProjectionDefinitionBuilder::new(name.into(), query)
    }
}

pub struct ProjectionDefinitionBuilder { /* ... */ }
impl ProjectionDefinitionBuilder {
    pub fn new(name: String, query: crate::query::TagFilter) -> Self;
    /// Add a scalar field with its event handlers.
    pub fn scalar(mut self, name: impl Into<String>, spec: ScalarFieldSpec) -> Self;
    /// Add a nested object field.
    pub fn object(mut self, name: impl Into<String>, spec: ObjectFieldSpec) -> Self;
    pub fn build(self) -> ProjectionDefinition;
}

pub struct ScalarFieldSpecBuilder { /* ... */ }
impl ScalarFieldSpecBuilder {
    pub fn new() -> Self;
    pub fn required(mut self) -> Self;
    pub fn default_value(mut self, v: Value) -> Self;
    pub fn on(mut self, event_type: impl Into<String>, handler: HandlerSpec) -> Self;
    pub fn build(self) -> ScalarFieldSpec;
}

pub struct ObjectFieldSpecBuilder { /* ... */ }
impl ObjectFieldSpecBuilder {
    pub fn new() -> Self;
    pub fn cleared_by(mut self, event_type: impl Into<String>) -> Self;
    pub fn field(mut self, name: impl Into<String>, spec: ScalarFieldSpec) -> Self;
    pub fn build(self) -> ObjectFieldSpec;
}

    Create `event-sourcing/src/projection/definition.rs` with exactly the types specified in the `<interfaces>` block. Key implementation notes:

    1. `HandlerSpec` uses `#[serde(rename_all = "snake_case")]` so variants serialize as `{"from":"$.x"}`, `{"value":null}`, `{"increment":1.0}`, etc. The enum is `#[serde(deny_unknown_fields)]`.

    2. `FieldSpec` uses `#[serde(untagged)]` — the deserializer tries `Scalar` first; if the JSON has `"type": "object"` it falls through to `ObjectFieldSpec`. This works because `ObjectFieldSpec` has a required `field_type` field that scalars don't have. Add `#[serde(deny_unknown_fields)]` on each variant struct (NOT on the untagged enum itself — `deny_unknown_fields` on untagged enums is unsupported in serde and will panic at runtime).

    3. `ObjectFieldSpec.field_type` field is renamed `type` in JSON via `#[serde(rename = "type")]`. Its value must always be the string `"object"`. On deserialization, serde will populate this field — callers should not rely on it being validated by serde; the engine validates it.

    4. `IndexMap<String, HandlerSpec>` and `IndexMap<String, FieldSpec>` and `IndexMap<String, ScalarFieldSpec>` must derive Serialize/Deserialize — this works automatically with `indexmap`'s `serde` feature.

    5. Builder structs: `ProjectionDefinitionBuilder`, `ScalarFieldSpecBuilder`, `ObjectFieldSpecBuilder`. Each uses the builder pattern: fields accumulate via method chaining, `build()` returns the target struct. All methods take `mut self` and return `Self` for chaining.

    6. Add `#[cfg(test)] mod tests` containing:
       - `handler_from_roundtrip`: `HandlerSpec::From { from: "$.amount".into() }` → JSON `{"from":"$.amount"}` → back, assert equality.

       - `projection_definition_deny_unknown_fields`: `serde_json::from_str::<ProjectionDefinition>(r#"{"name":"X","query":{"Equals":"a"},"fields":{},"extra":true}"#)` returns `Err`.
       - `scalar_deny_unknown`: `serde_json::from_str::<ScalarFieldSpec>(r#"{"events":{},"bogus":1}"#)` returns `Err`.
       - `insertion_order_preserved`: Build a ProjectionDefinition with fields added in order `["z", "a", "m"]`. Serialize, deserialize, check field key order is still `["z", "a", "m"]` (IndexMap preserves insertion order).

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04.02-01 | Tampering | ProjectionDefinition JSON deserialization | mitigate | `#[serde(deny_unknown_fields)]` on ProjectionDefinition, ScalarFieldSpec, ObjectFieldSpec, and HandlerSpec variants rejects unknown keys. Tested by `projection_definition_deny_unknown_fields` and `scalar_deny_unknown`. |
| T-04.02-02 | Tampering | HandlerSpec::Value with malicious JSON Value | accept | HandlerSpec::Value accepts any serde_json::Value — this is by design (literal values). The engine validates field types against EventSchemaDef at projection registration time (Plan 03), not at definition parse time. |
| T-04.02-03 | DoS | IndexMap with unbounded field count | accept | Field count is unbounded in Phase 4; application-level limits on definition size are the caller's responsibility. Documented limitation. |
</threat_model>
