```rust
// Generated struct
#[derive(Debug, serde::Deserialize)]
pub struct CustomerView {
    pub name: String,
    pub accountant_name: Option<String>,
    pub address: Option<CustomerViewAddress>,
}

#[derive(Debug, serde::Deserialize)]
pub struct CustomerViewAddress {
    pub city: Option<String>,
    pub postal_code: Option<String>,
}

// Generated ReadModel impl
impl event_sourcing::ReadModel for CustomerView {
    fn definition() -> event_sourcing::ProjectionDefinition {
        event_sourcing::ProjectionDefinition::builder(
            "CustomerView",
            event_sourcing::query::TagFilter::StartsWith("customer:".to_string()),
        )
        .scalar("name",
            event_sourcing::ScalarFieldSpecBuilder::new()
                .required()
                .on("CustomerRegistered", event_sourcing::HandlerSpec::From { from: "$.name".to_string() })
                .on("CustomerRenamed", event_sourcing::HandlerSpec::From { from: "$.name".to_string() })
                .build()
        )
        .scalar("accountant_name",
            event_sourcing::ScalarFieldSpecBuilder::new()
                .on("AccountantAssigned", event_sourcing::HandlerSpec::From { from: "$.name".to_string() })
                .on("AccountantRemoved", event_sourcing::HandlerSpec::Value { value: serde_json::Value::Null })
                .build()
        )
        .object("address",
            event_sourcing::ObjectFieldSpecBuilder::new()
                .cleared_by("AddressCleared")
                .field("city",
                    event_sourcing::ScalarFieldSpecBuilder::new()
                        .on("CustomerRegistered", event_sourcing::HandlerSpec::From { from: "$.city".to_string() })
                        .on("AddressChanged", event_sourcing::HandlerSpec::From { from: "$.city".to_string() })
                        .build()
                )
                .field("postal_code",
                    event_sourcing::ScalarFieldSpecBuilder::new()
                        .on("CustomerRegistered", event_sourcing::HandlerSpec::From { from: "$.postal_code".to_string() })
                        .on("AddressChanged", event_sourcing::HandlerSpec::From { from: "$.postal_code".to_string() })
                        .build()
                )
                .build()
        )
        .build()
    }
}
```

DSL parsing rules:
- `projection Name { ... }` → struct name = `Name`
- `query tag.starts_with("prefix:") as alias` → `TagFilter::StartsWith("prefix:")`
- `query tag.ends_with(":suffix") as alias` → `TagFilter::EndsWith(":suffix")`
- `query tag.equals("exact") as alias` → `TagFilter::Equals(Tag::new("exact"))`
- `field_name: alias.EventType.path | alias.Event2.path2` → scalar, required, From handlers
- `field_name?: alias.EventType.path | alias.Event2.path2` → scalar, optional, From handlers
- `field_name?: alias.EventType.path | alias.Other = null` → `Value { value: Null }` handler
- `field_name?: alias.EventType.path | alias.Other = "literal"` → `Value { value: String("literal") }` handler
- `field_name |+ alias.EventType.path` → `IncrementBy { increment_by: "$.path" }`
- `field_name |- alias.EventType.path` → `DecrementBy { decrement_by: "$.path" }`
- `field { ... } cleared_by alias.EventType` → object field with cleared_by
- `field? { ... }` → optional object
- The alias is stripped from generated JSON (primary stream alias is DSL-only per D-34)
- struct field type inference: required field → `String` if handlers use `$.path` on non-numeric event; `f64` for increment/decrement fields; `Option<T>` for `?` fields. For Phase 4, default all fields to `String` / `Option<String>` unless `|+`/`|-` operators are used (→ `f64`/`Option<f64>`).
- Nested object type: generate a companion struct named `{ParentStruct}{FieldNamePascalCase}`.

    The parser entry point:
    ```rust
    struct ProjectionInput {
        name: syn::Ident,
        alias: syn::Ident,
        query: QuerySpec,
        fields: Vec<FieldDecl>,
    }

    enum QuerySpec {
        StartsWith(String),
        EndsWith(String),
        Equals(String),
    }

    enum FieldDecl {
        Scalar {
            name: syn::Ident,
            optional: bool,
            handlers: Vec<HandlerDecl>,
        },
        Object {
            name: syn::Ident,
            optional: bool,
            fields: Vec<ScalarHandlerDecl>,
            cleared_by: Option<String>, // event type name
        },
    }

    struct HandlerDecl {
        alias: String,
        event_type: String,
        operation: OperationDecl,
    }

    enum OperationDecl {
        FromPath(String),        // alias.EventType.field → "$.field"
        ValueNull,               // = null
        ValueString(String),     // = "literal"
        IncrementBy(String),     // |+ alias.EventType.path → "$.path"
        DecrementBy(String),     // |- alias.EventType.path → "$.path"
    }
    ```
