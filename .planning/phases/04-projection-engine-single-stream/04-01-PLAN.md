    ```rust
    //! Event schema primitives. Defines FieldType (macro-generated enum per D-01/D-02),
    //! FieldDef, EventSchemaDef (per D-04), and the Event trait (per D-03).

    use std::collections::HashSet;

    #[derive(Debug, Clone, PartialEq, Eq, Hash, serde::Serialize, serde::Deserialize)]
    pub enum FieldType {
        String,
        Integer,
        Decimal,
        Date,
        DateTime,
    }

    #[derive(Debug, Clone, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
    #[serde(deny_unknown_fields)]
    pub struct FieldDef {
        pub name: String,
        pub field_type: FieldType,
    }

    #[derive(Debug, Clone, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
    #[serde(deny_unknown_fields)]
    pub struct EventSchemaDef {
        pub event_type: String,
        pub fields: Vec<FieldDef>,
    }

    impl EventSchemaDef {
        /// Strict compatibility (D-07): event_type must match AND field set must match
        /// (name + field_type pairs). Field ordering is ignored.
        pub fn is_compatible_with(&self, other: &EventSchemaDef) -> bool {
            if self.event_type != other.event_type {
                return false;
            }
            let a: HashSet<(&str, &FieldType)> =
                self.fields.iter().map(|f| (f.name.as_str(), &f.field_type)).collect();
            let b: HashSet<(&str, &FieldType)> =
                other.fields.iter().map(|f| (f.name.as_str(), &f.field_type)).collect();
            a == b
        }
    }

    pub trait Event {
        fn event_type() -> &'static str;
        fn schema() -> EventSchemaDef;
    }
    ```
